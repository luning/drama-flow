    ```python
    class TestDramasPersonalized:
        """AC-DRAMA-07/08/09: Personalized recommendation based on watch history"""

        def test_logged_in_personalized_sort(self, client, seed_user, seed_drama_fantasy2, auth_header):
            """AC-DRAMA-07 / REC-01: Logged-in user with watch history gets same-category-first sort.
            
            Setup: User has watched drama 1 (fantasy, category_id=4) via a watch record.
            Drama 6 (fantasy, category_id=4) is unwatched — same category.
            Drama 2 (president, category_id=5) is unwatched — different category.
            
            Expected: Drama 6 (same-category unwatched, priority 0) appears before drama 2 (other-category unwatched, priority 1).
            """
            # Create a watch record for drama 1 (fantasy category) — episode 1 belongs to drama 1
            client.put("/api/watch-records/1",
                       json={"progress": 50.0, "last_position": 300.0, "completed": False},
                       headers=auth_header)
            
            # Fetch personalized list
            response = client.get("/api/dramas", headers=auth_header)
            assert response.status_code == 200
            items = response.json()["items"]
            item_ids = [d["id"] for d in items]
            
            # Find positions
            pos_drama6 = item_ids.index(6)  # 龙族崛起 (fantasy, same category)
            pos_drama2 = item_ids.index(2)  # 霸道总裁爱上我 (president, other category)
            
            # Same-category unwatched (priority 0) must come before other-category unwatched (priority 1)
            assert pos_drama6 < pos_drama2, \
                f"Drama 6 (same category) at position {pos_drama6} should be before Drama 2 (other category) at position {pos_drama2}"
            
            # Also verify total is correct (5 existing + 1 new)
            assert response.json()["total"] == 6

        def test_completed_dramas_downgraded(self, client, seed_user, seed_drama_fantasy2, auth_header):
            """AC-DRAMA-08 / REC-02: Completed dramas appear at lower positions than in-progress.
            
            Setup: User has completed drama 3 (suspense) via completed=True watch record.
            User has in-progress drama 2 (president) via completed=False watch record.
            
            Expected: Drama 2 (in-progress, priority 2) appears before drama 3 (completed, priority 3).
            """
            # Create an in-progress record for drama 2 (episode 11 is drama 2's first episode)
            client.put("/api/watch-records/11",
                       json={"progress": 50.0, "last_position": 300.0, "completed": False},
                       headers=auth_header)
            
            # Create a completed record for drama 3 (episode 21 is drama 3's first episode)
            client.put("/api/watch-records/21",
                       json={"progress": 100.0, "last_position": 600.0, "completed": True},
                       headers=auth_header)
            
            # Fetch personalized list
            response = client.get("/api/dramas", headers=auth_header)
            assert response.status_code == 200
            items = response.json()["items"]
            item_ids = [d["id"] for d in items]
            
            pos_drama2 = item_ids.index(2)  # in-progress (priority 2)
            pos_drama3 = item_ids.index(3)  # completed (priority 3)
            
            assert pos_drama2 < pos_drama3, \
                f"In-progress Drama 2 at position {pos_drama2} should be before completed Drama 3 at position {pos_drama3}"

        def test_unauthenticated_default_sort(self, client, seed_drama_fantasy2):
            """AC-DRAMA-09 / REC-03: Unauthenticated user gets default sort (no personalization).
            
            Setup: No auth header, no user context.
            
            Expected: Same response structure as before, default sort (updated_at DESC).
            Total includes all 6 dramas.
            """
            response = client.get("/api/dramas")
            assert response.status_code == 200
            data = response.json()
            assert data["total"] == 6
            assert len(data["items"]) == 6
            # Response retains standard structure
            assert "page" in data
            assert "size" in data
            # Order is updated_at DESC (default) — verify dramas are returned without crash
            all_ids = [d["id"] for d in data["items"]]
            assert len(all_ids) == 6

        def test_no_watch_records_fallback(self, client, seed_user, seed_drama_fantasy2, auth_header):
            """REC-04: Authenticated user with NO watch records gets default sort (fallback).
            
            Setup: User is authenticated but has never watched anything.
            
            Expected: Default sort (same behavior as unauthenticated), not a personalized sort.
            """
            response = client.get("/api/dramas", headers=auth_header)
            assert response.status_code == 200
            data = response.json()
            assert data["total"] == 6
            assert len(data["items"]) == 6
            # Should not crash — personalized sort falls back to default when no records exist
            all_ids = [d["id"] for d in data["items"]]
            assert len(all_ids) == 6
    ```
